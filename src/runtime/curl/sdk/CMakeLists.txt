cmake_minimum_required(VERSION 3.22)
project(BunCurlSDK LANGUAGES C)
find_package(Perl REQUIRED)
# CURL_UPSTREAM_DIR is vendor/curl: the curl/curl archive scripts/build/deps/curl.ts pins and fetches.
if(NOT CURL_UPSTREAM_DIR OR NOT EXISTS "${CURL_UPSTREAM_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "CURL_UPSTREAM_DIR must name the curl source fetched by scripts/build/deps/curl.ts")
endif()
set(BUILD_CURL_EXE ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_STATIC_CURL ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CURL_WERROR OFF CACHE BOOL "" FORCE)
set(ENABLE_UNICODE ON CACHE BOOL "" FORCE)
set(ENABLE_CURL_MANUAL ON CACHE BOOL "" FORCE)
add_subdirectory("${CURL_UPSTREAM_DIR}" "upstream" EXCLUDE_FROM_ALL)
get_target_property(tool_sources curl SOURCES)
set(embedded_sources "")
foreach(source IN LISTS tool_sources)
  # curl.exe's version resource would land in the static library as a non-object member; bun.exe has its own.
  if(source MATCHES "\\.(rc|manifest)$")
    continue()
  endif()
  if(IS_ABSOLUTE "${source}")
    list(APPEND embedded_sources "${source}")
  else()
    if(source STREQUAL "tool_hugehelp.c" OR source STREQUAL "tool_ca_embed.c")
      list(APPEND embedded_sources "${CMAKE_BINARY_DIR}/upstream/src/${source}")
    else()
      list(APPEND embedded_sources "${CURL_UPSTREAM_DIR}/src/${source}")
    endif()
  endif()
endforeach()
add_library(BunCurlTool STATIC ${embedded_sources})
get_target_property(tool_includes curl INCLUDE_DIRECTORIES)
get_target_property(tool_definitions curl COMPILE_DEFINITIONS)
target_include_directories(BunCurlTool PRIVATE ${tool_includes} "${ZLIB_INCLUDE_DIR}")
target_compile_definitions(BunCurlTool PRIVATE ${tool_definitions} HAVE_CONFIG_H main=BunCurlMain wmain=BunCurlMain)
if(WIN32)
  target_compile_definitions(BunCurlTool PRIVATE UNICODE _UNICODE)
endif()
target_link_libraries(BunCurlTool PRIVATE libcurl_static)
set_target_properties(BunCurlTool PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set_target_properties(libcurl_static PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" OUTPUT_NAME "curl" PREFIX "${CMAKE_STATIC_LIBRARY_PREFIX}" DEBUG_POSTFIX "")
